cmake_minimum_required(VERSION 3.16)
project(NovaNative VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(nova-native src/nova_native.cpp)
if(MSVC)
  target_compile_options(nova-native PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(nova-native PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(nova-native PRIVATE user32 gdi32 shell32 comdlg32 winhttp)
  # Windows GUI-subsystem runtime; its main entry point is wWinMain.
  # Bytecode compiler supports import ui; interactive Win32 smoke is pending.
  add_executable(nova-windowed WIN32 src/nova_native.cpp)
  target_compile_definitions(nova-windowed PRIVATE NOVA_PORTABLE_GUI)
  if(MSVC)
    target_compile_options(nova-windowed PRIVATE /W4 /EHsc /utf-8)
  else()
    target_compile_options(nova-windowed PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  target_link_libraries(nova-windowed PRIVATE user32 gdi32 shell32 comdlg32 winhttp)
  add_executable(NovaStudio WIN32 studio/nova_studio.cpp)
  if(MSVC)
    target_compile_options(NovaStudio PRIVATE /W4 /EHsc /utf-8)
  else()
    target_compile_options(NovaStudio PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  target_link_libraries(NovaStudio PRIVATE user32 gdi32 comdlg32 shell32)
  install(TARGETS nova-native nova-windowed NovaStudio RUNTIME DESTINATION bin)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBCURL REQUIRED IMPORTED_TARGET libcurl)
  target_link_libraries(nova-native PRIVATE PkgConfig::LIBCURL ${CMAKE_DL_LIBS})
  install(TARGETS nova-native RUNTIME DESTINATION bin)
endif()

# Real native tests for the portable text engine used by Nova Studio (all OSes).
include(CTest)
if(BUILD_TESTING)
  add_executable(nova-studio-text-tests tests/studio_text_native.cpp)
  add_test(NAME nova-studio-text-tests COMMAND nova-studio-text-tests)
endif()
